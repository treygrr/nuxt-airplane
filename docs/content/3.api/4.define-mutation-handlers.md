```ts
function defineMutationHandlers(handlers: MutationHandlers): MutationHandlers
function defineMutationHandler<N extends MutationName>(name: N, handler: MutationHandler<N>): MutationHandlers
```

Both give you typed keys and typed `variables`; the singular form declares one mutation by name, for one-file-per-mutation layouts, and returns a map you can spread into the plural form. Default-export the result from a file in the [`handlersDir`](/getting-started/configuration#handlersdir).

## Types

```ts
type MutationHandlers = { [N in MutationName]?: MutationHandler<N> }

interface MutationHandler<N extends MutationName> {
  /** Summary for the pending list; computed once, before `apply`. */
  describe?: string | ((ctx: HandlerContext<MutationVariables<N>>) => string | Promise<string>)
  /** Runs when queued, inside an undo-logged transaction. */
  apply?: (ctx: ApplyContext<MutationVariables<N>>) => unknown
  /** Runs with the server response after the mutation was sent. */
  reconcile?: (ctx: ReconcileContext<MutationVariables<N>>, result: MutationResult<N>) => unknown
  /** Replaces the automatic undo on rollback. */
  revert?: (ctx: HandlerContext<MutationVariables<N>>) => unknown
  /** Queries to refetch after sync. Defaults to every mounted query. */
  refetch?: QueryName[]
}

interface HandlerContext<V> {
  db: AirplaneDexie
  cache: CacheApi
  variables: V
}

interface ApplyContext<V> extends HandlerContext<V> {
  /** A placeholder id for a row the server will create; stable across re-applies. */
  tempId: () => string
}

interface ReconcileContext<V> extends HandlerContext<V> {
  /** The ids `apply` minted, in call order. */
  tempIds: string[]
  /** Map a placeholder to the server's id; queued mutations are rewritten before they are sent. */
  resolve: (tempId: string, id: string) => Promise<void>
}

interface CacheApi {
  update<N extends QueryName>(name: N, updater: Updater<QueryResult<N>>): Promise<void>
  update<N extends QueryName>(name: N, variables: QueryVariables<N>, updater: Updater<QueryResult<N>>): Promise<void>
  get<N extends QueryName>(name: N, variables?: QueryVariables<N>): Promise<QueryResult<N> | undefined>
  set<N extends QueryName>(name: N, variables: QueryVariables<N>, data: QueryResult<N>): Promise<void>
  invalidate(name: QueryName): Promise<void>
}

type Updater<T> = (data: T, variables: unknown) => T | undefined | void
```

Hooks may return anything, so a Dexie call can be the whole handler. See [Mutation handlers](/guide/mutation-handlers) for when each hook runs and [Creating offline, then editing](/guide/mutations#creating-offline-then-editing) for `tempId` and `resolve`.

## Example

```ts [app/airplane/sites.ts]
export default defineMutationHandlers({
  // Against your own tables
  DeleteSite: {
    describe: async ({ db, variables }) => `Delete ${(await db.Site.get(variables.id))?.name}`,
    async apply({ db, variables }) {
      await db.Site.delete(variables.id)
      await db.SiteItem.where('id').anyOf(variables.itemIds).delete()
    },
    refetch: ['GetSites']
  },
  // Against a cached response
  ArchiveSite: {
    apply: ({ cache, variables }) =>
      cache.update('GetSites', (data) => {
        const site = data.sites.find(s => s.id === variables.id)
        if (site) site.archived = true
      })
  }
})
```
