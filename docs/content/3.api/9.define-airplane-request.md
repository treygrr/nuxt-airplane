```ts
function defineAirplaneRequest(options: AirplaneRequestOptions): AirplaneRequestOptions
```

Default-export it from a file in the [`handlersDir`](/getting-started/configuration#handlersdir). Several files are merged; the last one wins per option.

## Types

```ts
interface AirplaneRequestOptions {
  /** Object, or a per-request function run inside the Nuxt app context. */
  headers?: HeadersInit | ((ctx: AirplaneRequestContext) => HeadersInit | undefined | Promise<HeadersInit | undefined>)
  /** fetch credentials mode, e.g. 'include'. */
  credentials?: RequestCredentials
  /** A fetch implementation to use instead of the global one. */
  fetch?: typeof fetch
  /** Sees every response before it is parsed. */
  onResponse?: (response: Response, ctx: AirplaneRequestContext) => unknown
}

interface AirplaneRequestContext {
  document: string
  kind: 'query' | 'mutation' | 'subscription'
  variables: unknown
}
```

See [Auth and headers](/guide/auth-and-headers).

## Example

```ts [app/airplane/request.ts]
export default defineAirplaneRequest({
  headers: async () => ({ Authorization: `Bearer ${await getFreshToken()}` }),
  credentials: 'include',
  onResponse: (res) => { if (res.status === 401) navigateTo('/login') },
})
```
