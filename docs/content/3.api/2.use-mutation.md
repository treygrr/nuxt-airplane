---
title: useMutation
description: Apply a mutation locally and queue it for the server.
navigation:
  icon: i-lucide-pencil
---

```ts
function useMutation<N extends MutationName>(
  name: N,
  options?: { queue?: true }
): UseMutationReturn<MutationVariables<N>, MutationRow>

function useMutation<N extends MutationName>(
  name: N,
  options: { queue: false }
): UseMutationReturn<MutationVariables<N>, MutationResult<N>>
```

## Parameters

| | |
| --- | --- |
| `name` | A mutation document name from `NuxtAirplaneMutations`. |
| `options.queue` | `false` sends the mutation directly instead of queuing it; see [Online-only mutations](/guide/mutations#online-only-mutations). Default `true`. |

## Returns

```ts
interface UseMutationReturn<V, R> {
  mutate: (variables: V) => Promise<R>
  loading: Ref<boolean>
  error: Ref<Error | null>
}

interface MutationRow {
  id: number
  document: string
  variables: unknown
  /** From the handler's `describe`, or the humanized document name. */
  description: string
  status: 'pending' | 'syncing' | 'failed'
  createdAt: number
  error?: string
}
```

`mutate` resolves once the change is applied locally and queued; see [Mutations](/guide/mutations). It rejects only if the `apply` handler throws or the composable is called during SSR.

With `queue: false`, `mutate` resolves with the mutation's result type once the server has answered, `loading` is `true` while the request is in flight, and it rejects while offline or when the server returns errors.

## Example

```vue
<script setup lang="ts">
const setOnHand = useMutation('UpdateSiteItemOnHand')

function onChange(siteId: string, itemId: string, onHand: number) {
  setOnHand.mutate({ siteId, itemId, onHand })
}
</script>
```

Direct, for a mutation whose response the UI needs:

```vue
<script setup lang="ts">
const checkout = useMutation('Checkout', { queue: false })

async function pay() {
  const { checkout: order } = await checkout.mutate({ cartId })
  await navigateTo(`/orders/${order.id}`)
}
</script>
```
