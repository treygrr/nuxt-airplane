```ts
function useSync(): UseSyncReturn
```

## Returns

```ts
interface UseSyncReturn {
  online: Ref<boolean>
  networkOnline: Ref<boolean>
  forcedOffline: Ref<boolean>
  syncing: Ref<boolean>
  synced: ComputedRef<boolean>
  pending: Ref<MutationRow[]>
  syncTotal: Ref<number>
  syncCount: Ref<number>
  lastError: Ref<string | null>
  sync: () => Promise<void>
  rollback: (id: number) => Promise<void>
  retry: (id: number) => Promise<void>
  goOffline: () => void
  goOnline: () => void
}
```

See [Sync and offline](/guide/sync-and-offline) for the meaning of each field. During SSR every ref is a static placeholder (`online: true`, `synced: true`, `pending: []`) and the functions do nothing.

## Example

```vue
<script setup lang="ts">
const { pending, rollback, retry, goOffline, goOnline, forcedOffline } = useSync()
</script>

<template>
  <button v-if="forcedOffline" @click="goOnline()">Go online</button>
  <button v-else @click="goOffline()">Go offline</button>

  <ul>
    <li v-for="m in pending" :key="m.id">
      {{ m.description }} <em>{{ m.status }}</em>
      <button v-if="m.status === 'failed'" @click="retry(m.id)">Retry</button>
      <button :disabled="m.status === 'syncing'" @click="rollback(m.id)">Roll back</button>
    </li>
  </ul>
</template>
```
